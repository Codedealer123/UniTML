cmake_minimum_required(VERSION 3.14)
project(unitml VERSION 1.2.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(MSVC)
    add_compile_options(/W4 /WX-)
else()
    add_compile_options(-Wall -Wextra)
endif()

include_directories(${CMAKE_SOURCE_DIR}/third_party)

set(SOURCES
    src/main.c
    src/platform_single_instance.c
    src/deep_link.c
    src/health.c
    src/discovery.c
)

# --- ADD WEBVIEW.CC HERE ---
# Treat webview.cc as C++
set_source_files_properties(third_party/webview.cc PROPERTIES LANGUAGE CXX)
# Add it to the executable
list(APPEND SOURCES third_party/webview.cc)

if(WIN32)
    add_executable(unitml WIN32 ${SOURCES})
    target_link_libraries(unitml PRIVATE user32 ole32 shell32 gdi32 advapi32 ws2_32 winhttp)
    
    find_package(CURL QUIET)
    if(CURL_FOUND)
        target_link_libraries(unitml PRIVATE CURL::libcurl)
    else()
        message(STATUS "libcurl not found, will use dynamic loading")
    endif()
    
elseif(APPLE)
    add_executable(unitml MACOSX_BUNDLE ${SOURCES})
    
    # Link C++ standard library for webview.cc
    target_link_libraries(unitml PRIVATE stdc++)

    find_library(COCOA_LIBRARY Cocoa REQUIRED)
    find_library(WEBKIT_LIBRARY WebKit REQUIRED)
    target_link_libraries(unitml PRIVATE ${COCOA_LIBRARY} ${WEBKIT_LIBRARY} curl)
    
    set_target_properties(unitml PROPERTIES
        MACOSX_BUNDLE_BUNDLE_NAME "UniTML"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.unitml.app"
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
    )
    
elseif(UNIX)
    add_executable(unitml ${SOURCES})
    
    # Link C++ standard library for webview.cc
    target_link_libraries(unitml PRIVATE stdc++)

    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEBKIT2 webkit2gtk-4.0)
    if(NOT WEBKIT2_FOUND)
        pkg_check_modules(WEBKIT2 webkit2gtk-4.1)
    endif()
    if(NOT WEBKIT2_FOUND)
        message(FATAL_ERROR "WebKitGTK not found (tried webkit2gtk-4.0 and webkit2gtk-4.1)")
    endif()
    pkg_check_modules(CURL REQUIRED libcurl)
    
    target_include_directories(unitml PRIVATE ${WEBKIT2_INCLUDE_DIRS} ${CURL_INCLUDE_DIRS})
    target_link_libraries(unitml PRIVATE ${WEBKIT2_LIBRARIES} ${CURL_LIBRARIES})
    target_compile_options(unitml PRIVATE ${WEBKIT2_CFLAGS_OTHER} ${CURL_CFLAGS_OTHER})
endif()

install(TARGETS unitml DESTINATION bin)
